cmake_minimum_required(VERSION 3.19)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
project(korri_webos_viewer LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_RPATH ON)
option(KV_HOST_TEST "Workstation dummy media, never ship to TV" OFF)

# Sources are immutable Nix inputs; common-c is the same checked-in fork as Android.
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
# Use the kernel CSPRNG in both our client and common-c's mbedTLS entropy pool.
add_compile_definitions(MBEDTLS_NO_PLATFORM_ENTROPY MBEDTLS_ENTROPY_HARDWARE_ALT)
add_subdirectory("${MBEDTLS_SOURCE}" mbedtls EXCLUDE_FROM_ALL)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
add_subdirectory("${COMMON_SOURCE}" common EXCLUDE_FROM_ALL)
target_include_directories(moonlight-common-c PRIVATE "${MBEDTLS_SOURCE}/include")
target_link_libraries(moonlight-common-c PRIVATE mbedcrypto)

set(EXPAT_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(EXPAT_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(EXPAT_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(EXPAT_WITH_GETRANDOM OFF CACHE BOOL "" FORCE)

add_subdirectory("${EXPAT_SOURCE}" expat EXCLUDE_FROM_ALL)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_DISABLE_INTRINSICS ON CACHE BOOL "" FORCE)
set(OPUS_FIXED_POINT ON CACHE BOOL "" FORCE)
set(OPUS_ENABLE_FLOAT_API OFF CACHE BOOL "" FORCE)
set(OPUS_STACK_PROTECTOR OFF CACHE BOOL "" FORCE)
add_subdirectory("${OPUS_SOURCE}" opus EXCLUDE_FROM_ALL)

# Build only the fixed media backends. No SDL, Moonlight UI or power/input hooks.
set(CORE "${SS4S_SOURCE}/src")
add_library(ss4s SHARED "${CORE}/library.c" "${CORE}/driver.c" "${CORE}/player.c"
    "${CORE}/audio.c" "${CORE}/video.c" "${CORE}/module.c" "${CORE}/stats.c"
    "${CORE}/module_libdl.c" "${CORE}/mutex_pthread.c")
target_include_directories(ss4s PUBLIC "${SS4S_SOURCE}/include"
    PRIVATE "${SS4S_SOURCE}/modules/interface/include" "${CORE}")
target_compile_definitions(ss4s PRIVATE SS4S_LIBRARY_SUFFIX=".so")
target_compile_options(ss4s PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ss4s PRIVATE dl pthread rt)
target_link_options(ss4s PRIVATE -Wl,--no-undefined)
function(media_module name)
    target_include_directories(${name} PRIVATE "${SS4S_SOURCE}/include"
        "${SS4S_SOURCE}/modules/interface/include" "${SS4S_SOURCE}/modules/webos/utils")
    set_target_properties(${name} PROPERTIES PREFIX "")
    target_link_options(${name} PRIVATE -Wl,--no-undefined)
endfunction()
if(KV_HOST_TEST)
    set(DUMMY "${SS4S_SOURCE}/modules/dummy")
    add_library(ss4s-dummy SHARED "${DUMMY}/dummy_module.c" "${DUMMY}/dummy_player.c"
        "${DUMMY}/dummy_video.c" "${DUMMY}/dummy_audio.c")
    media_module(ss4s-dummy)
    target_link_libraries(ss4s-dummy PRIVATE pthread)
    install(TARGETS ss4s-dummy LIBRARY DESTINATION lib)
else()
    find_path(NDL_INCLUDE NAMES NDL_directmedia.h PATH_SUFFIXES libndl-media REQUIRED)
    find_library(NDL_LIBRARY NAMES NDL_directmedia REQUIRED)
    set(NDL "${SS4S_SOURCE}/modules/webos/ndl/webos4")
    add_library(ss4s-ndl-webos4 SHARED "${NDL}/ndl_module.c" "${NDL}/ndl_player.c"
        "${NDL}/ndl_video.c" "${NDL}/ndl_audio.c"
        "${SS4S_SOURCE}/modules/webos/utils/jail_check.c"
        "${SS4S_SOURCE}/modules/webos/utils/m3_kadp_fix.c"
        "${SS4S_SOURCE}/modules/webos/utils/max_res.c")
    media_module(ss4s-ndl-webos4)
    target_include_directories(ss4s-ndl-webos4 PRIVATE "${NDL_INCLUDE}")
    target_compile_definitions(ss4s-ndl-webos4 PRIVATE _GNU_SOURCE)
    target_link_libraries(ss4s-ndl-webos4 PRIVATE "${NDL_LIBRARY}" pthread dl)
    find_path(PULSE_INCLUDE NAMES pulse/simple.h REQUIRED)
    find_library(PULSE_SIMPLE NAMES pulse-simple REQUIRED)
    find_library(PULSE NAMES pulse REQUIRED)
    add_library(ss4s-pulse SHARED "${SS4S_SOURCE}/modules/pulse/pulse.c")
    media_module(ss4s-pulse)
    target_include_directories(ss4s-pulse PRIVATE "${PULSE_INCLUDE}")
    target_link_libraries(ss4s-pulse PRIVATE "${PULSE_SIMPLE}" "${PULSE}")
    install(TARGETS ss4s-ndl-webos4 ss4s-pulse LIBRARY DESTINATION lib)
endif()
add_library(korri-webos-viewer STATIC identity.c https.c protocol.c stream.c input.c)
target_include_directories(korri-webos-viewer PUBLIC . "${COMMON_SOURCE}/src"
    PRIVATE "${MBEDTLS_SOURCE}/include" "${OPUS_SOURCE}/include")
target_link_libraries(korri-webos-viewer PUBLIC moonlight-common-c mbedtls mbedx509 mbedcrypto expat opus ss4s pthread rt m)
target_compile_options(korri-webos-viewer PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
add_executable(webos-viewer-test cli.c)
target_link_libraries(webos-viewer-test PRIVATE korri-webos-viewer)
target_compile_options(webos-viewer-test PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
if(KV_HOST_TEST)
    target_compile_definitions(webos-viewer-test PRIVATE KV_HOST_TEST)
    add_executable(viewer-protocol-test protocol_test.c)
    target_link_libraries(viewer-protocol-test PRIVATE korri-webos-viewer)
    target_compile_options(viewer-protocol-test PRIVATE -Wall -Wextra -Werror)
endif()
install(TARGETS korri-webos-viewer moonlight-common-c enet mbedtls mbedx509 mbedcrypto opus expat ss4s webos-viewer-test
    ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES viewer.h "${COMMON_SOURCE}/src/Limelight.h" DESTINATION include/korri-webos-viewer)

# The TV LS2 role recognizes this exact executable, not a second viewer ELF.
if(KORRI_WEBOS_LAUNCHER)
    add_library(korri-media-test OBJECT "${KORRI_MEDIA_SOURCE}")
    target_include_directories(korri-media-test PRIVATE "${SS4S_SOURCE}/include")
    if(KV_HOST_TEST)
        target_compile_definitions(korri-media-test PRIVATE main=korri_media_test_main MEDIA_DRIVER="dummy")
    else()
        target_compile_definitions(korri-media-test PRIVATE main=korri_media_test_main MEDIA_DRIVER="ndl-webos4")
    endif()
    add_executable(korri-webos "${KORRI_WEBOS_LAUNCHER}" mode.c $<TARGET_OBJECTS:korri-media-test>)
    target_compile_definitions(korri-webos PRIVATE KORRI_MEDIA_TEST KORRI_VIEWER)
    target_compile_options(korri-webos PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow)
    target_link_libraries(korri-webos PRIVATE korri-webos-viewer)
    set(CMAKE_SKIP_RPATH OFF)
    set_target_properties(korri-webos PROPERTIES SKIP_BUILD_RPATH TRUE INSTALL_RPATH "$ORIGIN/../lib")
    set_target_properties(ss4s PROPERTIES SKIP_BUILD_RPATH TRUE INSTALL_RPATH "$ORIGIN")
    install(TARGETS korri-webos RUNTIME DESTINATION bin)
endif()
