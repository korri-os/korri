set pagination off
set confirm off
set print thread-events off
handle all nostop noprint pass
set $before = 0
set $after = 0
set $destroyed = 0
break eglMakeCurrent
commands
  silent
  printf "== [gdb thread %d] eglMakeCurrent(display=%p, context=%p)\n", $_gthread, $rdi, $rcx
  printf "   current context before the call:\n"
  print ((void*(*)(void)) eglGetCurrentContext)()
  thread
  bt 7
  continue
end
break eglDestroyContext
commands
  silent
  printf "== [gdb thread %d] eglDestroyContext(display=%p, context=%p)\n", $_gthread, $rdi, $rsi
  thread
  bt 7
  set $destroyed = 1
  continue
end
break eglCreateImage
commands
  silent
  if $destroyed == 0 && $before < 2
    set $before = $before + 1
    printf "== [gdb thread %d] eglCreateImage before the swap; current context:\n", $_gthread
    print ((void*(*)(void)) eglGetCurrentContext)()
  end
  if $destroyed == 1 && $after < 3
    set $after = $after + 1
    printf "== [gdb thread %d] eglCreateImage after the destroy; current context:\n", $_gthread
    print ((void*(*)(void)) eglGetCurrentContext)()
  end
  continue
end
continue
info threads
detach
quit
